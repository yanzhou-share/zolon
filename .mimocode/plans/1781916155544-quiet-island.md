# 文档表格处理方案

## 问题分析

当前 `parse_docx` 只提取段落，表格内容完全丢失：
```python
def parse_docx(file_path: str) -> str:
    from docx import Document
    doc = Document(file_path)
    return "\n".join([para.text for para in doc.paragraphs if para.text.strip()])
```

## 解决方案

### 方案 1：表格转文本（推荐）

将表格转换为结构化文本：

```python
def parse_docx(file_path: str) -> str:
    from docx import Document
    doc = Document(file_path)
    parts = []
    
    for element in doc.element.body:
        if element.tag.endswith('}p'):
            # 段落
            text = element.text
            if text and text.strip():
                parts.append(text)
        elif element.tag.endswith('}tbl'):
            # 表格
            table = Table(element, doc)
            parts.append(table_to_text(table))
    
    return "\n".join(parts)


def table_to_text(table) -> str:
    """将表格转换为易读文本"""
    rows = []
    for row in table.rows:
        cells = [cell.text.strip() for cell in row.cells]
        rows.append(cells)
    
    if not rows:
        return ""
    
    # 方式1：Markdown 表格格式
    header = rows[0]
    lines = ["| " + " | ".join(header) + " |"]
    lines.append("| " + " | ".join(["---"] * len(header)) + " |")
    for row in rows[1:]:
        lines.append("| " + " | ".join(row) + " |")
    
    return "\n".join(lines)


def table_to_kv(table) -> str:
    """将两列表格转换为键值对格式"""
    rows = []
    for row in table.rows:
        cells = [cell.text.strip() for cell in row.cells]
        rows.append(cells)
    
    if len(rows) < 2:
        return ""
    
    # 第一行作为标题，后续行作为数据
    parts = []
    for row in rows[1:]:
        if len(row) >= 2:
            parts.append(f"{row[0]}: {row[1]}")
    
    return "\n".join(parts)
```

### 方案 2：PDF 表格提取

使用 `pdfplumber` 提取表格：

```python
def parse_pdf_tables(file_path: str) -> str:
    import pdfplumber
    
    text_parts = []
    with pdfplumber.open(file_path) as pdf:
        for page in pdf.pages:
            # 提取普通文本
            page_text = page.extract_text()
            if page_text:
                text_parts.append(page_text)
            
            # 提取表格
            tables = page.extract_tables()
            for table in tables:
                text_parts.append(table_to_text_simple(table))
    
    return "\n".join(text_parts)


def table_to_text_simple(table) -> str:
    """简单的表格转文本"""
    if not table:
        return ""
    lines = []
    for row in table:
        if row:
            lines.append(" | ".join(str(cell) if cell else "" for cell in row))
    return "\n".join(lines)
```

## 智能表格格式选择

根据表格结构选择最佳格式：

```python
def format_table(table) -> str:
    """智能选择表格格式"""
    rows = []
    for row in table.rows:
        cells = [cell.text.strip() for cell in row.cells]
        rows.append(cells)
    
    if not rows:
        return ""
    
    # 检查是否为两列键值对表格（如规格参数）
    if len(rows[0]) == 2 and all(len(r) == 2 for r in rows):
        return table_to_kv(table)
    
    # 检查是否为多列数据表格
    if len(rows[0]) > 2:
        return table_to_markdown(table)
    
    # 默认格式
    return table_to_plain(table)
```
